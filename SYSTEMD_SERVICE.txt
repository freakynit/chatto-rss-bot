# sudo mkdir -p /var/log/chatto-rss-bot
# Install Java 21 and Maven, and install the sibling chatto-java-sdk with mvn install.
# Verify the mvn and JAVA_HOME paths below on the VPS before starting.
# Configure config.yaml, including the public callback URL and webhook port/path.
# This bot gets its installs from Bot Hub callbacks (stored in feeds.sqlite
# in the working directory). Register the bot in the hub dashboard
# and install it after this service starts; until then it waits for the callback.
# One instance serves installs on any number of servers. No .env file is used.

[Unit]
Description=Chatto RSS Bot
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=root
Group=root

WorkingDirectory=/opt/codes/chatto-rss-bot

# Verify with: which mvn
ExecStart=/usr/bin/mvn exec:java

Environment=JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64

Restart=always
RestartSec=5

KillSignal=SIGTERM
TimeoutStopSec=30

# Send logs to systemd's journal.
# StandardOutput=journal
# StandardError=journal

# Send logs to our directory; create it before starting the service.
StandardOutput=append:/var/log/chatto-rss-bot/out.log
StandardError=append:/var/log/chatto-rss-bot/error.log

NoNewPrivileges=true
PrivateTmp=true

[Install]
WantedBy=multi-user.target
