[Unit]
Description=Chatto Bot Java Service
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=root
Group=root

WorkingDirectory=/opt/codes/chatto-rss-bot

# Verify with: which mvn
ExecStart=/usr/bin/mvn exec:java

Environment=JAVA_HOME=/usr/lib/jvm/java-21-openjdk-amd64

Restart=always
RestartSec=5

KillSignal=SIGTERM
TimeoutStopSec=30

# Send logs to systemd's journal.
# StandardOutput=journal
# StandardError=journal

# Send logs to our directory
StandardOutput=append:/var/log/chatto-bot/out.log
StandardError=append:/var/log/chatto-bot/error.log

NoNewPrivileges=true
PrivateTmp=true

[Install]
WantedBy=multi-user.target
